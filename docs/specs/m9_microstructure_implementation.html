1. Architecture Overview

1.1 Module Structure

agent/quant/microstructure.py
├── SpreadEstimator (nn.Module)
│   ├── corwin_schultz(high, low) → spread
│   └── roll_measure(returns) → spread
├── OrderFlowImbalance (nn.Module)
│   ├── compute_ofi_trade_tape(prices, volumes, timestamps)
│   └── compute_ofi_zscore(ofi, window)
├── VPINCalculator (nn.Module)
│   ├── bucket_trades(trades, bucket_volume)
│   ├── classify_buckets_bvc(buckets)
│   └── compute_vpin(buckets, n_buckets)
├── KyleLambdaEstimator
│   └── estimate_lambda(prices, signed_volume, window_hours)
└── MicrostructureFeatureExtractor (nn.Module)
    └── forward(instrument_data) → feature_dict

1.2 Design Constraints

Differentiability Invariant: All forward passes must be in PyTorch. Non-differentiable operations (OLS regression for Kyle's λ) are precomputed offline and stored as static features. Only normalization/aggregation in forward pass.
Signal Depth L2: All features are per-instrument (entity-level), not market-wide aggregates. Cross-entity features (L3) deferred to future phase.

2. Component Specifications

2.1 SpreadEstimator (M9.4)

2.1.1 Corwin-Schultz Method

class SpreadEstimator(nn.Module):
    def corwin_schultz(
        self,
        high: torch.Tensor,  # shape: (T,) daily highs
        low: torch.Tensor,   # shape: (T,) daily lows
        variant: str = "CSD"  # "CSD" (daily censored) | "CSM" | "CSP"
    ) -> torch.Tensor:
        """
        Compute Corwin-Schultz spread estimator.
        
        Returns: (T-1,) tensor of daily spreads
        
        Algorithm:
        1. β = [ln(H_t / L_t)]²
        2. γ = [ln(H_{t,t+1} / L_{t,t+1})]²
        3. α = (√(2β) - √β) / (3 - 2√2) - √(γ / (3 - 2√2))
        4. S = 2(e^α - 1) / (1 + e^α)
        5. Apply censoring per variant
        """

Numerical Stability:

  • Clamp log arguments to min=1e-8 to avoid log(0)
  • Clamp α to [-10, 10] before exp to avoid overflow
  • Set NaN spreads to 0.0

2.1.2 Roll Measure

def roll_measure(
    self,
    returns: torch.Tensor  # shape: (T,) log returns
) -> torch.Tensor:
    """
    Compute Roll (1984) spread from return autocovariance.
    
    Returns: scalar spread estimate
    
    Algorithm:
    1. Compute Cov(r_t, r_{t-1})
    2. If Cov < 0: Spread = 2√(-Cov)
    3. Else: Spread = 0
    """

2.2 OrderFlowImbalance (M9.1)

class OrderFlowImbalance(nn.Module):
    def compute_ofi_trade_tape(
        self,
        prices: torch.Tensor,    # (T,) trade prices
        volumes: torch.Tensor,   # (T,) trade volumes
        timestamps: torch.Tensor # (T,) unix timestamps
    ) -> torch.Tensor:
        """
        Compute trade-tape OFI using BVC-style price change classification.
        
        Returns: (T,) OFI values
        
        Algorithm (per research, simplified):
        1. For each window [t-Δt, t]:
           - Aggregate volume
           - Compute price change: Δp = p_close - p_open
           - Sign volume: V_buy ~ Φ(Δp / σ), V_sell ~ 1 - V_buy
        2. OFI = V_buy - V_sell
        """
        
    def compute_ofi_zscore(
        self,
        ofi: torch.Tensor,  # (T,) raw OFI
        window: int = 720   # 1-hour lookback (720 5-sec bars)
    ) -> torch.Tensor:
        """
        Standardize OFI to z-score vs. trailing distribution.
        
        Returns: (T,) z-scored OFI
        """

2.3 VPINCalculator (M9.2)

class VPINCalculator(nn.Module):
    def __init__(self, bucket_volume: float = 100000.0, n_buckets: int = 50):
        super().__init__()
        self.bucket_volume = bucket_volume
        self.n_buckets = n_buckets
        
    @dataclass
    class VolumeBucket:
        open_price: float
        close_price: float
        high: float
        low: float
        total_volume: float
        timestamp_end: int
        
    def bucket_trades(
        self,
        trades: List[Tuple[float, float, int]]  # (price, volume, timestamp)
    ) -> List[VolumeBucket]:
        """
        Partition trades into equal-volume buckets.
        
        Algorithm:
        1. Accumulate trades until total_volume >= bucket_volume
        2. Record bucket OHLC
        3. Start new bucket
        """
        
    def classify_buckets_bvc(
        self,
        buckets: List[VolumeBucket]
    ) -> torch.Tensor:
        """
        Classify each bucket volume as buy/sell using BVC.
        
        Returns: (N, 2) tensor [[V_buy_1, V_sell_1], ...]
        
        Algorithm:
        1. Δp = (close - open) / bucket volatility estimate
        2. V_buy = V_total × Φ(Δp)  # normal CDF
        3. V_sell = V_total - V_buy
        """
        
    def compute_vpin(
        self,
        classified_buckets: torch.Tensor  # (N, 2) [V_buy, V_sell]
    ) -> torch.Tensor:
        """
        Compute VPIN over rolling window.
        
        Returns: (N - n_buckets + 1,) VPIN values
        
        Algorithm:
        VPIN = (1/n) × Σ|V_buy_i - V_sell_i| / V_bucket
        """

2.4 KyleLambdaEstimator (M9.5)

class KyleLambdaEstimator:
    """
    Rolling OLS estimation of price impact.
    
    NOTE: Not a nn.Module — OLS is non-differentiable.
    This is a feature generator, run offline.
    """
    
    def estimate_lambda(
        self,
        prices: np.ndarray,         # (T,) hourly prices
        signed_volume: np.ndarray,  # (T,) hourly net buy volume
        window_hours: int = 720     # 30 days
    ) -> np.ndarray:
        """
        Rolling OLS: ΔP_t = α + λ × Q_t + ε_t
        
        Returns: (T - window_hours + 1,) λ estimates
        
        Algorithm:
        1. For each day T:
           - Window: [T - window_hours, T]
           - ΔP = np.diff(prices)
           - Q = signed_volume[:-1]
           - OLS: λ̂ = (Q' Q)^{-1} Q' ΔP
        2. Store λ̂_T
        """

2.5 MicrostructureFeatureExtractor

class MicrostructureFeatureExtractor(nn.Module):
    """
    Unified extractor that computes all M9 features for one instrument.
    """
    
    def __init__(self):
        super().__init__()
        self.spread_estimator = SpreadEstimator()
        self.ofi_calculator = OrderFlowImbalance()
        self.vpin_calculator = VPINCalculator()
        # Kyle's λ is precomputed offline
        
    def forward(
        self,
        ohlcv: Dict[str, torch.Tensor],  # daily OHLC + volume
        trades: Optional[List] = None,    # tick data if available
        precomputed_lambda: Optional[torch.Tensor] = None
    ) -> Dict[str, torch.Tensor]:
        """
        Extract all available M9 features.
        
        Returns:
        {
            "spread_cs": (T,),
            "spread_roll": (T,),
            "ofi": (T,),
            "ofi_zscore": (T,),
            "vpin": (T,),
            "vpin_regime": (T, 3),  # one-hot [normal, elevated, extreme]
            "kyle_lambda": (T,),
            "lambda_regime": (T, 3)  # one-hot [liquid, normal, illiquid]
        }
        """

3. Graph Builder Integration

3.1 New Method in GraphBuilder

# In agent/models/gnn/graph_builder.py

class GraphBuilder:
    def __init__(self, ...):
        ...
        self.microstructure_extractor = MicrostructureFeatureExtractor()
        
    def add_microstructure_features(
        self,
        graph: HeteroData,
        instrument_ohlcv: Dict[str, pd.DataFrame],
        precomputed_lambda: Optional[Dict[str, torch.Tensor]] = None
    ) -> HeteroData:
        """
        Add M9 features to instrument nodes.
        
        Args:
            graph: Existing HeteroData graph
            instrument_ohlcv: {instrument_id: DataFrame with OHLCV}
            precomputed_lambda: {instrument_id: λ time series}
            
        Returns:
            graph with updated instrument node features
        """
        micro_features = []
        
        for inst_id in graph["instrument"].node_id:
            ohlcv = instrument_ohlcv[inst_id]
            lambda_ts = precomputed_lambda.get(inst_id) if precomputed_lambda else None
            
            feats = self.microstructure_extractor(
                ohlcv=ohlcv,
                precomputed_lambda=lambda_ts
            )
            
            # Flatten to single vector
            micro_vec = torch.cat([
                feats["spread_cs"][-1:],     # most recent day
                feats["spread_roll"][-1:],
                feats["ofi_zscore"][-1:],    # most recent window
                feats["vpin"][-1:],
                feats["vpin_regime"][-1],    # (3,)
                feats["kyle_lambda"][-1:],
                feats["lambda_regime"][-1]   # (3,)
            ])  # total: 1+1+1+1+3+1+3 = 11 dims
            
            micro_features.append(micro_vec)
            
        # Concat with existing instrument features
        micro_tensor = torch.stack(micro_features)  # (N_instruments, 11)
        graph["instrument"].x = torch.cat([
            graph["instrument"].x,
            micro_tensor
        ], dim=1)
        
        return graph
